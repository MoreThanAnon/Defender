🧪 Basic Example: Passport or Utility Bill Detection

Copy code
import pytesseract
from pdf2image import convert_from_path

# Set this path to your Tesseract install (Windows only)
# pytesseract.pytesseract.tesseract_cmd = r'C:\Program Files\Tesseract-OCR\tesseract.exe'

def extract_text_from_pdf(pdf_path):
    pages = convert_from_path(pdf_path)
    text = ""
    for page in pages:
        text += pytesseract.image_to_string(page)
    return text

def detect_document_type(text):
    if "passport" in text.lower() or "mrz" in text:
        return "Likely Passport"
    elif any(word in text.lower() for word in ["utility bill", "account number", "billing", "electricity", "gas"]):
        return "Likely Utility Bill"
    else:
        return "Unknown Document Type"

pdf_path = "example.pdf"
text = extract_text_from_pdf(pdf_path)
result = detect_document_type(text)
print(result)
